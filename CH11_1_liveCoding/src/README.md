class Book {
    string title;   // 책 제목
    int isbn;       // ISBN 번호
    double price;   // 가격
    bool isNovel;   // 소설 책이니?

public:
    Book(string t = "", int i = 0, double p = 0, bool n = false) {
        title = t; isbn = i; price = p; isNovel = n;
    }
    ostream& print(ostream& out) const {
        return out << title << " " << isbn << " " << price << " " << isNovel;
    }
    void println() const { print(cout) << endl; }
};
