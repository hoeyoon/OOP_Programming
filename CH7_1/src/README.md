/******************************************************************************
 * [문제 4] Person의 p2 = p 연산자 추가
 ******************************************************************************/
// 1) 기존 Person 클래스에 아래 대입 연산자 = 를 구현하라. 
//----------------------------------------------------------------------------
    Person& operator = (const Person& p);
//----------------------------------------------------------------------------
// 즉, p2 = p; 객체 p의 모든 멤버 데이타를 객체 p2에 대입하는 연산자이다.
// 이전에 구현했던 Person의 멤버 함수 assign()을 = 연산자로 대체하면 쉽게 구현할 수 있다.
// 즉, assign()이라는 멤버 함수 이름을 없애고 대신 = 연산자로 교체하라. 함수가 하는 일은 동일하다.
// = 연산자는 객체 자신의 참조자를 반환한다. 즉, *this 를 반환하면 된다.
//----------------------------------------------------------------------------
// 위 = 연산자 구현 시 [주의]해야 할 것은 일반 멤버들은 = 를 사용하여 각 멤버를 바로 대입하면 된다.
// 그러나 포인터 멤버들(address, memo)은 먼저 할당 받은 메모리를 반납한 후 새로 메모리를 할당 받고 
// 복사해야 한다. 이를 위해 setAddress()를 호출해야지 copyAddress() 바로 호출하면 안된다.
//----------------------------------------------------------------------------
// 2) 기존의 CopyConstructor 클래스 내에서 assign() 함수를 호출하던 곳을 찾아 = 연산자로 교체하라.
//----------------------------------------------------------------------------
// 참고로 Person p = p1;과 같은 문장의 =는 Person p(p1);와 같이 복사생성자를 이용해 
// 기존 객체 p1을 복사해서 새로 생성되는 객체 p를 초기화시키는 문장으로 대체되며 
// 대입 연산자 =와는 다른 개념이다. 대입 연산자 =는 p = p2; 문장처럼 이미 초기화된 객체 p를 
// 다른 p2의 값으로 객체 내용을 변경하는 것을 의미한다.  
//----------------------------------------------------------------------------
// 3) 아래 함수를 OperatorOverload 클래스에 추가하고 run() 함수에서 호출할 수 있도록 하라.
//----------------------------------------------------------------------------

    Person& operator++();    // 객체 자신에 대한 참조자 리턴
    Person  operator++(int); // 기존 객체(this)를 복사하여 새로운 지역 객체를 만들고  
           // 기존 객체의 체중을 1 증가시킨다. 그리고 새로운 지역 객체(기존 복사본)를 반환한다.

//----------------------------------------------------------------------------
// 아래 함수를 OperatorOverload 클래스에 추가하고 run() 함수에서 호출할 수 있도록 하라.
//----------------------------------------------------------------------------

    void personIncrement() { // Memu item 5
        Person p1(p); p1.setAddress("");
        cout << "p1  : "; p1.println();
        cout << "++p1: "; (++p1).println(); // operator ++()
        cout << "p1++: "; (p1++).println(); // operator ++(int)
        cout << "p1  : "; p1.println();
        cout << "p2 = p1++" << endl;
        Person p2 = p1++; // copy constructor
        cout << "p2  : "; p2.println();
        cout << "p1  : "; p1.println();
        cout << "p2 = (++p1)++" << endl;
        p2 = (++p1)++;
        cout << "p2  : "; p2.println();
        cout << "p1  : "; p1.println();
        cout << "p2 = ++(p1++)" << endl;
        p2 = ++(p1++);
        cout << "p2  : "; p2.println();
        cout << "p1  : "; p1.println();
        cout << "p2 = ++p1++" << endl;
        p2 = ++p1++;
        cout << "p2  : "; p2.println();
        cout << "p1  : "; p1.println();
        cout << "(p2 = 2.0 + ++p1++ + 3.0) == p : "
             << ((p2 = 2.0 + ++p1++ + 3.0) == p) << endl;
        cout << "p2  : "; p2.println();
        cout << "p1  : "; p1.println();
    }

// 2) 아래 함수를 OperatorOverload 클래스에 추가하고 run() 함수에서 호출할 수 있도록 하라.

    void typeConversion() { // Memu item 7
        Person p1(p), p2(p);
        p1 << "Hong" << 1; // 이름과 id를 객체 p1에 설정함
        p2 << "Dong" << 2;
        cout << "p1: "; p1.println();
        cout << "p2: "; p2.println();

        // 아래 문장은 int id = (int)p1; 과 동일; 컴파일러가 id와 동일한 타입인 int로 
        // 타입을 변경하는 operator int()를 자동으로 호출하여 리턴 값을 id에 저장
        int id = p1;

        // 아래 문장은 (string)p1 와 동일; 컴파일러가 name과 동일한 타입인 string로 
        // 타입을 변경하는 operator string()를 자동으로 호출하여 리턴 값을 name에 저장
        string name = p1;

        cout << "p1 name:" << name << ", id:" << id << endl;
        p2 >> id >> name; // 객체 p2로부터 id와 이름을 얻어옴
        cout << "p2 name:" << name << ", id:" << id << endl;

        // 아래 함수호출 시 컴파일러가 알아서 자동으로 함수 매개변수와 동일한 타입으로 변환해 주는 
        // operator string()과  operator int()를 호출하여 리턴 값을 함수인자로 넘겨 줌 
        print_name_id(p1, p1); 
        print_name_id(p2, p2);

        // 아래 문장은 사용자가 출력할 데이타 타입을 정확히 지정해 주면 
        // 컴파일러가 상응하는 operator string()과  operator int()를 호출한다.
        cout << "p2 name:" << (string)p2 << ", id:" << (int)p2 << endl;
        cout << "p1+p2 name:" << (string)p1+name << ", id:" << (int)p1+id << endl;
    }
